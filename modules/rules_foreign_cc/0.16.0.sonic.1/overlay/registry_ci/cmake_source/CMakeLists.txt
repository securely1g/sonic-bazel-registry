cmake_minimum_required(VERSION 3.15)
project(header_path_regression LANGUAGES CXX)
find_path(SDK_HEADERS sdk_warning.h REQUIRED)
find_path(SYSTEM_HEADERS system_warning.h REQUIRED)
find_path(ORDINARY_HEADERS ordinary.h REQUIRED)
add_library(header_paths OBJECT paths.cc)
target_include_directories(header_paths PRIVATE ${SDK_HEADERS} ${SYSTEM_HEADERS} ${ORDINARY_HEADERS})
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/receipt.txt" "CMake: staged system and ordinary headers retained\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/receipt.txt" DESTINATION .)
